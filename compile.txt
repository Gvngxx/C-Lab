====================================================================
                COMPILACION DEL PROYECTO C++ CON OPENGL
====================================================================

REQUISITOS:
- MinGW64 (g++, gcc, cmake, mingw32-make)
- GLFW debe compilarse primero

====================================================================
PASO 1: COMPILAR GLFW (hace una sola vez)
====================================================================

cd lib\glfw
mkdir build
cd build

..\..\..\mingw64\bin\cmake.exe .. -G "MinGW Makefiles" -DCMAKE_C_COMPILER=C:\workspace\C++LAB\mingw64\bin\gcc.exe -DCMAKE_CXX_COMPILER=C:\workspace\C++LAB\mingw64\bin\g++.exe -DCMAKE_MAKE_PROGRAM=C:\workspace\C++LAB\mingw64\bin\mingw32-make.exe -DGLFW_BUILD_EXAMPLES=OFF -DGLFW_BUILD_TESTS=OFF -DGLFW_BUILD_DOCS=OFF

..\..\..\mingw64\bin\mingw32-make.exe

====================================================================
PASO 2: COMPILAR EL PROYECTO (desde el directorio raiz)
====================================================================

cd c:\workspace\C++LAB

REM Eliminar ejecutable anterior
del /Q bin\CppLabExample.exe 2>nul || exit /b 0

REM Compilar con g++
mingw64\bin\g++.exe -o bin\CppLabExample.exe src\main.cpp lib\glad\src\gl.c -I lib\glad\include -I lib\glfw\include -L lib\glfw\build\src -lglfw3 -lopengl32 -lgdi32 -luser32 -lkernel32 -std=c++17

REM Verificar que se creo
dir bin\CppLabExample.exe

====================================================================
COMANDOS INDIVIDUALES (BASH SCRIPT)
====================================================================

# Compilar GLFW
cd lib/glfw && mkdir -p build && cd build
../../../mingw64/bin/cmake.exe .. -G "MinGW Makefiles" \
  -DCMAKE_C_COMPILER=C:/workspace/C++LAB/mingw64/bin/gcc.exe \
  -DCMAKE_CXX_COMPILER=C:/workspace/C++LAB/mingw64/bin/g++.exe \
  -DCMAKE_MAKE_PROGRAM=C:/workspace/C++LAB/mingw64/bin/mingw32-make.exe \
  -DGLFW_BUILD_EXAMPLES=OFF -DGLFW_BUILD_TESTS=OFF -DGLFW_BUILD_DOCS=OFF

../../../mingw64/bin/mingw32-make.exe

# Compilar proyecto principal
cd /d c:\workspace\C++LAB
del /Q bin\CppLabExample.exe 2>nul || exit /b 0
mingw64\bin\g++.exe -o bin\CppLabExample.exe src\main.cpp lib\glad\src\gl.c \
  -I lib\glad\include -I lib\glfw\include -L lib\glfw\build\src \
  -lglfw3 -lopengl32 -lgdi32 -luser32 -lkernel32 -std=c++17

COMANDO RAPIDO PARA COMPILAR (desde raiz del proyecto)
====================================================================

EJECUTAR EL PROGRAMA
====================================================================

bin\CppLabExample.exe

====================================================================
